Many processes will provide different results upon repitition with the same inputs, which we call randomness.
Statistics is the study of randomness and random expirements, but interestingly there is no good definition of randomness.
Instead, we simply say a random expirement is any process that exhibits this behavior, and do not concern ourselves with the particulars of what causes that randomness.

\begin{definition}
Let $S$ be a set. Say $\Omega \subseteq \mathcal{P}(S)$ is a $\sigma$-field whenever:
\begin{enumerate}
\item \begin{equation}
\emptyset \in \Omega
\end{equation}
\item \begin{equation} 
 \forall E \in \Omega, E^C \in \Omega
 \end{equation}
\item If $\{E_n\}$ is a countable collection in $\Omega$, 
\begin{equation}
\cup_n E_n \in \Omega
\end{equation}
\item If $\{E_n\}$ is a countable collection in $\Omega$, 
\begin{equation}
\cap_n E_n \in \Omega
\end{equation}
\end{enumerate}
\end{definition}

\textbf{Ex-} Let $S=\{1, 2, 3, 4\}$. THen $\{\emptyset, \{1,2\}, \{3,4\}, S\}$ is a $\sigma$-field.

\begin{lemma}
Let $S$ be a set and $\Omega_1$, $\Omega_2$ be $\sigma$-fields on $S$. Then $\Omega_1 \cap \Omega_2$ is a $\sigma$-field on $S$
\end{lemma}
\begin{proof}
Clearly, $\emptyset \in \Omega_1 \cap \Omega_2$, as $\emptyset \in \Omega_1, \Omega_2$.
Next, let $E\in \Omega_1 \cap \Omega_2$.
THen $E \in \Omega_1 , \Omega_2$, so $E^C \in \Omega_1 , \Omega_2$.
Thus, $E^C \in \Omega_1 \cap \Omega_2$. \\
Finally, let $\{E_n \}$ be a countable collection of sets in $\Omega_1 \cap \Omega_2$. 
THen every $E_n$ is in both $\Omega_1$ and $\Omega_2$.
So $\cup_n E_n, \cap_n E_n \in \Omega_1, \Omega_2$.
Thus, $\cap_n E_n, \cup_n E_n \in \Omega_1 \cap \Omega_2$.
Therefore, $\Omega_1 \cap \Omega_2$ is a $\sigma$-field.
\end{proof}

\begin{lemma}
Let $(S, \Omega, p)$ be a probability space and $E \in \Omega$.
Then 
\begin{equation}
p(E^C) = 1 - p(E)
\end{equation}
\end{lemma}
\begin{proof}
Notice $\{ E, E^C \}$ is a collection of pairwise disjoint sets in $\Omega$, so \\$p(E \cap E^C ) = p(E) + p(E^C )$.
Also, $E\cup E^C=S$, so $p(E) + p(E^C )= 1$.
\end{proof}

\begin{theorem}
\textbf{Principle of Inclusion-Exclusion:}
Let $(S, \Omega, p)$ be a probability space and $\{E_n\}$ be a collection of events.
Then
\begin{itemize}
\item \begin{equation}
p(E_1 \cup E_2 ) = p(E_1) + p(E_2) - p(E_1 \cap E_2 )
\end{equation}
\item \begin{multline}
p(E_1 \cup E_2 \cup E_3 ) = p(E_1) + p(E_2) + p(E_3)_ - p(E_1 \cap E_2 ) - p(E_1 \cap E_3) \\
- p(E_2 \cap E_3 ) + p(E_1 \cap E_2 \cap E_3 )
\end{multline}
\item and so on, where we add $(-1)^{k-1}$ times the sum of hte probabilities of all $k$-sized intersections of the events
\end{itemize}
\end{theorem}
\begin{proof}
Let $E$ and $F$ be any two events and consider that $E=(E \cap F ) \cup (E \cap F^C )$ and $F = (E \cap F) \cup (E^C \cap F)$.
So $E \cup F = (E \cap F) \cup (E \cap F^C ) \cup (E^C \cap F)$.
Also, $\{E \cup F, E \cup F^C, E^C \cup F\}$ is a countable collection of disjoint sets.
So $p(E \cup F ) =  p(E \cap F) + p(E \cap F^C ) + p(E^C \cap F)$ and $p(E) + p(F) = 2 p(E \cap F) + p(E \cap F^C ) + p(E^C \cap F)$.
So $p(E) + P(F) = p(E\cup F) + p(E\cap F)$.
Now, we will show the PIE by induction. \\
In the base case, let $n=2$ be the number of events and let $E=E_1$ and $F=E_2$. Then we have established $p(E_1) + p(E_2) = p(E_1 \cap E_2 ) + p(E_1 \cup E_2)$.
Thus, this establishes the base case. \\
Now, suppose the PIE holds for some $n \in \mathbb{N}$.
We will let $E=\cup_{k=1}^n E_k$ and $F=E_{n+1}$.
Then $p(\cup_{k=1}^{n+1} E_k ) = p(\cup_{k=1}^n E_k ) + p(E_{n+1}) - p((\cup_{k=1}^n E_k) \cap E_{n+1}) =p(\cup_{k=1}^n E_k ) + p(E_{n+1}) - p(\cup_{k=1}^n (E_k \cap E_{n+1}))$.
By applying PIE to the first and third terms, we have the desired result.
Therefore, by induction, PIE holds for all $N$.
\end{proof}

\begin{definition}
Say a collection of events $\{ E_n \}$ is a cover when $S = \cup_n E_n$.
\end{definition}

\begin{definition}
Say a partition is a cover when all $E_n$ are pairwise disjoint.

\begin{lemma}
If $\{E_n \} \searrow E$ then $p(E_n ) \searrow p(E)$.
\end{lemma}
\begin{proof}
Let $R_n = E_n \cap E_{n+1}^C$.
Then $\{R_k\}_{k=1}^\infty$ and $E$ together form a countable partition of $E_n$.
So $p(E_n) = p(E) + \sum_{k=n}^\infty p(R_k)$.
Because $p(E)$ and $p(E_n)$ are finite, the sum converges.
Let $\epsilon > 0$.
Then there is some $N \in \mathbb{N}$ such that for all $n \geq N$, $\sum_{k=n}^\infty p(R_k) \leq \epsilon$.
So $p(E) \leq p(E_n) \leq p(E) = \epsilon$.
\end{proof}
\begin{corrolary}
If $\{E_n\} \nearrow E$, $p(E_n) \nearrow p(E)$.
\end{corrolary}

\begin{theorem}
\textbf{Borel-Cantelli Lemma:}
Let $(S, \Omega, p)$ be a probability space, $\{ E_n \}$ a sequence of events, and $E = \limsup\limits_{n} E_n$.
If $\sum_{n=1}^\infty p(E_n) \leq \infty$ then $p(E) = 0$.
\end{theorem}
\begin{proof}
Recall $\limsup\limits_{n} E_n = \cap_{m \geq 1} \cup_{n \geq m} E_n$.
For all $m \geq 1$, $E \subseteq \cup_{n \geq m} p(E_n$.
Furthermore, this union induces a decreasing sequence of sets as $m$ increases.
So $p(E) \leq p(\cup_{n \geq m} E_n) \leq \sum_{n\geq m} p(E_n)$.
If $\sum_{n\geq 1} p(E_n) \leq \infty$ then $\sum_{n\geq m} p(E_n) \rightarrow 0$ as $m \rightarrow \infty$.
Thus, $p(E)=0$.
\end{proof}
\begin{corrolary}
If $\sum_{n=1}^\infty p(E_n) = \infty$ then $p(E) = 1$.
\end{corrolary}

\textbf{Note-} 
The Borel-Cantelli lemma is used to understand the probability of events that occur infinitely often. 
It is commonly used in random walks and actuarial analysis to predict the prevelance of certain events.

\begin{definition}
Let $E_1, E_2 \subseteq S$ be events. Say the conditional probability is 
\begin{equation}
p(E_1 | E_2) = \frac{p(E_1 \cap E_2 )}{p(E_2)}
\end{equation}
\end{definition}

\begin{theorem}
\textbf{Chain Rule:}
Let $(S, \Omega, p)$ be a sample space and $E_1, E_2, E_3$ be events.
\begin{equation}
p(E_1,E_2|E_3)=p(E_1|E_2,E_3)p(E_2|E_3)
\end{equation}
\end{theorem}
\begin{proof}
\begin{align*}
p(E_1| E_2 , E_3) p(E_2 | E_3) &= p(E_1 | E_2 \cap E_3) p(E_2 | E_3) \\
&= \frac{p(E_1 \cap E_2 \cap E_3 )}{p(E_2 \cap E_3)} \frac{p(E_2 \cap E_3)}{p(E_3)} \\
&= \frac{p(E_1 \cap E_2 \cap E_3)}{p(E_3)} \\
&= p(E_1 \cap E_2 | E_3) \\
&= p(E_1, E_2 | E_3)
\end{align*}
\end{proof}

\begin{theorem}
\textbf{Bayes' Theorem:}
Given partition $\{E_n \}$ and event $E$,
\begin{equation}
p(E_k | E) = \frac{p(E|E_k)p(E_k)}{\int_i p(E | E_i) dp(E_i)}
\end{equation}
\end{theorem}
\begin{proof}
Notice $p(E | E_k)p(E_k) = p(E \cap E_k)$.
Similarly, for any $i$, $p(E | E_i)p(E_i) = p(E \cap E_i)$.
Because $E_i$ form a partition of the sample space, $E_i \cap E$ forms a partition of $E$.
Thus,
\begin{align*}
\int_i p(E|E_i)dp(E_i) &= \sum_i p(E|E_i)dp(E_i) \\
&= \sum_i p(E_i \cap E) \\
&= p(\cup_i (E_i \cap E)) \\
&= p(E)
\end{align*}
From the definition of conditional probability, the ratio gives $p(E | E_k)$.
\end{proof}

\textbf{EX-} Suppose there are 2 urns.
One, labelled H, contains 3 red marbles and 2 blue marbles.
The other, labelled T, contains 4 red marbles and 2 blue marbles.
First, a coin is flipped, and an urn is selecrted at random for the corresponding urn.
What is the probability of heads if we draw a blue marble?
\begin{align*}
p(H|blue) &= \frac{p(blue|H)p(H)}{p(blue|H)p(H) + p(blue|T)p(T)} \\
&= \frac{(\frac{2}{5})(\frac{1}{2})}{(\frac{3}{5}) (\frac{1}{2}) + (\frac{2}{6})(\frac{1}{2})} \\
&= \frac{6}{11}
\end{align*}

\begin{definition}
Say 2 events $E_1, E_2$ in a probability space are independent whenever
\begin{equation}
p(E_1 | E_2) = p(E_1)
\end{equation}
\end{definition}

\begin{lemma}
Two events $E_1$ and $E_2$ are independent if and only if
\begin{equation}
p(E_1 \cap E_2) = p(E_1)p(E_2)
\end{equation}
\end{lemma}
\begin{proof}
First, suppose $p(E_1) = p(E_1 | E_2) = \frac{p(E_1 \cap E_2)}{p(E_2)}$.
So $p(E_1)p(E_2) = \\ p(E_1 \cap E_2)$. \\
Conversely, if $p(E_1 \cap E_2) = p(E_1)p(E_2)$, $p(E_1 | E_2) = \frac{p(E_1 \cap E_2)}{p(E_2)} \\= \frac{p(E_1)p(E_2)}{p(E_2)} = p(E_1)$.
\end{proof}

\begin{definition}
Let $(S, \Omega, p)$ be a probability space.
Say $p$ is a fair distribution whenever, for all events $E \in \Omega$ and all symmetries $g: S \rightarrow S$,
\begin{equation}
p(E) = p(g(E))
\end{equation}
\end{definition}

\textbf{Note-}
In all previous examples, we have assumed a fair distribution.

\begin{lemma}
If $(S, d)$ is a compact metric space, there is a fair distribution on $S$.
\end{lemma}
\begin{proof}
Let $\mathcal{B}$ be the borel $\sigma$-field on $S$ and $G$ be the group of symmetries acting on $S$.
For all $x, y \in S$, say $x \sim y$ whenever there is some $g \in G$ such that $y = g(x)$.
Notice $\sim$ is an equivalence relation.
For each equivalence class, select a representative element $x_i$.
Because $S$ is compact, every $[x_i]$ is also compact.
Thus, $[x_i]$ is isomorphic to a subgoup of $G$.
So $[x_i]$ is a topological group.
Let $\mu_i$ be teh Haar measure on $[x_i]$ with total measure 1. \\
Now, consider the quotient space $S/_\sim$will also be compact.
let $\nu$ be the Haar measure on this quotiont space with total measure 1.
Then, we can define a measure $\mu$ on $S$ as
\begin{equation*}
\mu(E) = \sum_i \mu_i(E \cap [x_i]) \nu([x_i])
\end{equation*}
Notice that $\mu$ has total measure 1, so it is a valid probability distribution. \\
Finally, consider that $g(E) = \cup_i g(E) \cap [x_i]= \cup_i g(E \cap [x_i])$.
so
\begin{align*}
\mu(g(E)) &= \sum_i \mu_i (g(E) \cap [x_i]) \nu([x_i]) \\
&= \sum_i \mu_i (g(E \cap [x_i])) \nu([x_i]) \\
&= \sum_i \mu_i (E \cap [x_i]) \nu([x_i]) \\
&= \mu(E)
\end{align*}
Thus, the distribution is fair.
\end{proof}
\begin{corrolary}
If $(S, d)$ is a compact metric space and the symmetry group on $S$ acts transitively, then there is a unique fair distribution on $S$.
\end{corrolary}

\textbf{Note`}
The lemma  states that there is a fair distirbution, but htere may be more than 1.
In ordewr for there to be a unique fair distribution, the symmetry group needds to act transitively on $S$.
This means the group $S/_\sim$ is trivial, and the fair distribution is tha Haar measure.
Without this, we can partition the sample space into fair subsets, but we have now way to ensure a consistent weighting via $\nu$ across that partition. \\

\textbf{Ex-} Consider $\mathbb{N}$.
This is not a compact metric space, so there is no guarontee there is a fair distibution on $\mathbb{N}$.
Indeed, suppose there were such a distribution.
Then $Pr(n) = p$ for some real number $p\geq 0$.
If $p>0$, $\sum_i p = \infty$.
Conversely, if $p=0$, then $\sum_i p = 0$.
In either case, this is not a valid probability distribution.
So no fair probability distribution can exist on $\mathbb{N}$.

\begin{definition}
\textbf{Monty Hall Problem:}
There are 3 doors, behind 2 of which are a goat and behind the third is a car.
You are given the option to select a door, and you receive the prize behind whichever door you select.
However, before you are given the prize, a different door you did not choose is opened to reveal a goat, and you are given the opportunity to switch doors to the other closed option.
What is the optimal strategy and what is the probability you receive the car when following it?
\end{definition}

\textbf{Ex-}
Suppose without loss of generality that you select door 1.
There are 3 options, which we will denote CGG, GCG, and GGC.
In the first 2 cases, door 3 is opened to reveal a goat, whereas in case 3 door 2 is opened to reveal a goat.
This means the options are now CGX, GCX, and GXC.
In 2 cases you do not have the car and in 1 case you do, so the optimal strategy is to switch, and the probability you get the car is $\frac{2}{3}$. \\

\textbf{Ex-}
In order to build intuition for the Monty Hall problem, instead consider a variation in which there are countably many doors, each labeled with a natural number.
You select door 1, and all doors except door 1 and 4328934 are opened to reveal goats.
In this case, it is obvious that you should switch, and you will almost surely receive the car.

\begin{definition}
\textbf{Bertrand's Paradox:}
Consider a unit circle and inscribe an equilateral triangle within it.
Randomly select a chord of the circle.
What is the probability the chord is longer than the side of the triangle?
\end{definition}

\textbf{Ex-}
We can uniquely determine a chord of a circle by its endpoints.
Without loss of generality, select 1 endoint to be a vertex of the triangle.
The triangle partitions the circle into 3 arcs of equal length.
If the other end of hte chord falls on one of the arcs adjacent to the chosen vertex, the chord will be shorter.
However, if the other end is on the opposite side the chord will be longer.
Thus, the probability is $\frac{1}{3}$.

\textbf{Ex-}
A chord can be uniquely determined by a radius and a point on that radius by taking hte chord perpindicular to the radius, intersecting it at that point.
The radius selected clearly does not affect the length of the chord, only the point slected, so without loss of generality select the perpindicular bisecot of one side of the triangle.
The side of the triangle passes through the radius at a distance of $\frac{1}{2}$ from the center, and if the point is closer to the center the chord will be longer.
Thus, the probability is $\frac{1}{2}$.

\textbf{Ex-}
A chord of a circle is uniquely determined by its midpoint.
Note that if the midpoint is inside the inscribed circle of hte triangle, the chord will be longer.
This inscribed circle has radius $\frac{1}{2}$, so the probability is the raio of the areas of the circles, or $\frac{1}{4}$.

\textbf{Note-}
The resolution of Bertrand's paradox is that the phrase "pick a random chord" is not a well-defined problem.
Even an attempt to pick one fairly is not defined.
This is because the group of symmetries on the set of chords does not act transitively.
Indeed, the symmetries of the situation are rotational symmetry by any angle, but rotating a chorc can not change its length.
Thus, a chord smaller than the triangle side can never be transformed into a chord longer than the triangle side.
While the sample space does not have transitive symmetry, it is compact, which is why we can create multiple reasonable definitions for a fair distribution over the chords.

\begin{definition}
\textbf{Sleepin Beauty Problem:}
On Sunday night, Sleeping Beauty is put to sleep and a fair coin is flipped.
If it comes up Heads, she will be awoken on Monday, interviewed, and a mild anesthetic will be administered that makes her forget the experience and puts her back to sleep.
If the coin comes up tails, this process happens on both Monday and Tuesday.
In both cases, she is awoken on Wednesday and the expirement ends.
During the interview(s), Sleeping Beauty is asked to what she believes the probability the coin came up heads.
Given she knows the expiremental set up, what should she say?
\end{definition}

\textbf{Ex-}
The coin is fair, so the probability of heads is $\frac{1}{2}$.
nothing in the experiment provides Sleepign Beauty any extra information, so she should say $\frac{1}{2}$. \\ 

\textbf{Ex-}
There are three situations: heads on Monday, tails on Monday, and tails on Tuesday.
These three situations are indistinguishable, so they should all have equal probability.
Thus, the probability of heads is $\frac{1}{3}$.
Indeed, when an expirement is repeated many times, the proportion of times an event occurs is its probability.
If Sleeping Beauty always assumes the coin came up heads, $\frac{1}{3}$ of the time she is interviewed she willb e correct.
This means we should interpret the probability to be $\frac{1}{3}$. \\ 

\textbf{Note-}
Monty Hall, Bertrand's Paradox, and the Sleeping Beauty problem are the three classic paradoxes of probability theory.
Monty Hall is an example of how probability can be counter-intuitive, especially conditional probability, and we must carefully consider the known information.
Bertrand's paradox shows that we must be careful about how we define probability distributions and ensure that a fair distribution is well-defined if that is our assumption.
The Sleeping Beauty paradox makes us question what to do when the observation of an event is not independent with the presence of an observer.
In this case, how should we account for the relationship between an observation and the observer?
